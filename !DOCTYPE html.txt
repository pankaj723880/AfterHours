  <textarea id="htmlCode" readonly></textarea>

  <script>
    let selectedRow = null;
    let ascending = true;

    function updateSourceCode() {
      const tbody = document.querySelector("#infoTable tbody");
      let html = "";
      tbody.querySelectorAll("tr").forEach(row => {
        const cells = row.querySelectorAll("td");
        if (cells.length > 3) {
          html += `  <tr>\n    <td>${cells[0].innerText}</td>\n    <td>${cells[1].innerText}</td>\n    <td>${cells[2].innerText}</td>\n    <td>\n      <button class='action-btn edit-btn' onclick='editRow(this)'>Edit</button>\n      <button class='action-btn delete-btn' onclick='deleteRow(this)'>Delete</button>\n    </td>\n  </tr>\n`;
        }
      });
      document.getElementById("htmlCode").value = html;
    }

    function addRow() {
      const name = document.getElementById("nameInput").value.trim();
      const age = document.getElementById("ageInput").value.trim();
      const city = document.getElementById("cityInput").value.trim();

      if (!name || !age || !city) {
        alert("Please fill all fields!");
        return;
      }

      if (selectedRow == null) {
        insertRow(name, age, city);
      } else {
        selectedRow.cells[0].innerText = name;
        selectedRow.cells[1].innerText = age;
        selectedRow.cells[2].innerText = city;
        selectedRow = null;
        document.getElementById("submitBtn").textContent = "Add Row";
      }

      clearInputs();
      filterTable();
      updateSourceCode();
    }

    function insertRow(name, age, city) {
      const table = document.getElementById("infoTable").getElementsByTagName('tbody')[0];
      const newRow = table.insertRow();
      newRow.innerHTML = `
        <td>${name}</td>
        <td>${age}</td>
        <td>${city}</td>
        <td>
          <button class="action-btn edit-btn" onclick="editRow(this)">Edit</button>
          <button class="action-btn delete-btn" onclick="deleteRow(this)">Delete</button>
        </td>
      `;
    }

    function editRow(button) {
      selectedRow = button.parentElement.parentElement;
      document.getElementById("nameInput").value = selectedRow.cells[0].innerText;
      document.getElementById("ageInput").value = selectedRow.cells[1].innerText;
      document.getElementById("cityInput").value = selectedRow.cells[2].innerText;
      document.getElementById("submitBtn").textContent = "Update Row";
    }

    function deleteRow(button) {
      if (confirm("Are you sure you want to delete this row?")) {
        const row = button.parentElement.parentElement;
        row.remove();
        if (selectedRow === row) clearInputs();
        filterTable();
        updateSourceCode();
      }
    }

    function filterTable() {
      const nameFilter = document.getElementById("nameFilter").value.toLowerCase();
      const ageFilter = document.getElementById("ageFilter").value.toLowerCase();
      const cityFilter = document.getElementById("cityFilter").value.toLowerCase();

      const rows = document.querySelectorAll("#infoTable tbody tr");

      rows.forEach(row => {
        const name = row.cells[0].innerText.toLowerCase();
        const age = row.cells[1].innerText.toLowerCase();
        const city = row.cells[2].innerText.toLowerCase();

        const matches = name.includes(nameFilter) && age.includes(ageFilter) && city.includes(cityFilter);
        row.style.display = matches ? "" : "none";
      });
    }

    function sortTable(columnIndex) {
      const tbody = document.querySelector("#infoTable tbody");
      const rows = Array.from(tbody.rows);
      rows.sort((a, b) => {
        const valA = a.cells[columnIndex].innerText.toLowerCase();
        const valB = b.cells[columnIndex].innerText.toLowerCase();
        return ascending ? valA.localeCompare(valB) : valB.localeCompare(valA);
      });
      ascending = !ascending;
      rows.forEach(row => tbody.appendChild(row));
      updateSourceCode();
    }

    function clearInputs() {
      document.getElementById("nameInput").value = "";
      document.getElementById("ageInput").value = "";
      document.getElementById("cityInput").value = "";
      document.getElementById("submitBtn").textContent = "Add Row";
      selectedRow = null;
    }

    function exportCSV() {
      const rows = Array.from(document.querySelectorAll("table tr"));
      const csv = rows.map(row => Array.from(row.cells).map(cell => cell.innerText).join(",")).join("\n");
      const blob = new Blob([csv], { type: 'text/csv' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = "table_data.csv";
      a.click();
    }

    // Initialize source code display on page load
    window.onload = updateSourceCode;
  </script>
